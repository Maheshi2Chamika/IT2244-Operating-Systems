(Q8)Arithmetic operators



#include <stdio.h>

double calculate(double num1, double num2, char operator1) {
    switch (operator1) {
        case '+': return num1 + num2;
        case '-': return num1 - num2;
        case '*': return num1 * num2;
        case '/':
            if (num2 == 0) {
                printf("\nError: Division by zero is not allowed!\n");
                return 0;
            }
            return num1 / num2;
        default:
            printf("\nError: Invalid operator. Use +, -, *, or /\n");
            return 0;
    }
}

int main() {
    char operator1, choice;
    double num1, num2, result;

    printf("===== Simple Calculator =====\n");

    do {

        printf("\nEnter the first number: ");
        if (scanf("%lf", &num1) != 1) {
            printf("\nError: Invalid input. Please enter a valid number.\n");
            return 1;
        }


        printf("Enter an operator (+, -, *, /): ");
        scanf(" %c", &operator1);

        printf("Enter the second number: ");
        if (scanf("%lf", &num2) != 1) {
            printf("\nError: Invalid input. Please enter a valid number.\n");
            return 1;
        }


        result = calculate(num1, num2, operator1);
        printf("\nResult: %.2lf\n", result);


        printf("\nDo you want to perform another calculation? (y/n): ");
        scanf(" %c", &choice);

    } while (choice == 'y' || choice == 'Y');

    printf("\nThank you for using the calculator! Goodbye.\n");

    return 0;
}
