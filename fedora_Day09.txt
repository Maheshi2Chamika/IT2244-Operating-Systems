(Q1)Fibonacci numbers

----------------------------------------------------
a=0
b=1
echo -n "$a $b "
for (( i=2; i<10; i++))
do
	fib=$((a+b ))
	echo -n "$fib "
	a=$b
	b=$fib
done

(Q2)Factorial
----------------------------------------------------
fact=1
num=5
for (( i=1; i<=num; i++))
do
	fact=$((fact*i))
done
echo "Factorial $num = $fact"

----------------------------------------------------

(Q3)Multiples of 3 between 1 and 50

----------------------------------------------------
#using while loop
let i=3
while [ $i -le 50 ];
do
	echo $i
	i=$(($i+3))
done

#using for loop
for (( i=3; i<=50; i+=3))
do
	echo $i;
done
