Q6)	output:-Enter two numbers:
			7 5
			7 is greater than 5
			
	output:-Enter two numbers:
			1 9
			1 is less than 9
			
**************************************************	
echo "Enter numbers"
read num1 num2
if [ "$num1" -gt "$num2" ]; then
    echo "$num1 is greater than $num2"
elif [ "$num1" -lt "$num2" ]; then
    echo "$num1 is less than $num2"
else
    echo "$num1 is equal to $num2"
fi
