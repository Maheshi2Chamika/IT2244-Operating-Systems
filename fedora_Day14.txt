(Q6)Write a c program to generate and print the Fibonacci series up to a specified number of terms.
The program should take the number of terms as input from the user and then 
display the corresponding Fibonacci sequence


#include <stdio.h>
int main() {
    int n, a = 0, b = 1, next = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    printf("Fibonacci series up to %d:\n", n);

    while (a <= n) {
        printf("%d ", a);
        next = a + b;
        a = b;
        b = next;
    }

    return 0;
}

(Q8)Write a c program that takes a binary number (as an integer) as input and converts it to its decimal equivalent.

#include <stdio.h>
int main() {
    int binary, decimal = 0, base = 1, remainder;

    printf("Enter a binary number: ");
    scanf("%d", &binary);

    while (binary > 0) {
        remainder = binary % 10;  // Extract the last digit
        decimal += remainder * base;  // Convert and add to decimal
        binary /= 10;  // Remove the last digit
        base *= 2;  // Update base (powers of 2)
    }

    printf("Decimal equivalent: %d\n", decimal);

    return 0;
}

(Q9)Write a c program that:
Accepts an array of integers from the user.
Finds and displays the maximum and minimum values in the array


#include <stdio.h>
int main() {
    int n, i, max, min;
    
    printf("Enter the number of elements: ");
    scanf("%d", &n);

    int arr[n]; // Declare array with user-defined size

    printf("Enter %d elements:\n", n);
    for(i = 0; i < n; i++) 
	{
        scanf("%d", &arr[i]);
    }

    max = min = arr[0]; // Initialize max and min with the first element

    for(i = 1; i < n; i++) 
	{
        if(arr[i] > max) 
		{
            max = arr[i]; 
        }
        if(arr[i] < min) 
		{
            min = arr[i];
        }
    }

    printf("Maximum value: %d\n", max);
    printf("Minimum value: %d\n", min);

    return 0;
}

(Q10)c program to generate Pascal's Triangle

Output:
no of rows =5
          1
        1   1
      1   2   1
    1   3   3   1
  1   4   6   4   1


#include <stdio.h>
int main() {
    int rows, coef = 1, space, i, j;
    
    printf("Enter number of rows: ");
    scanf("%d", &rows);
    
    for (i = 0; i < rows; i++) 
	{
        for (space = 1; space <= rows - i; space++)
            printf("  ");
        
        for (j = 0; j <= i; j++) 
		{
            if (j == 0 || i == 0)
                coef = 1;
            else
                coef = coef * (i - j + 1) / j;
            
            printf("%4d", coef);
        }
        printf("\n");
    }
    
    return 0;
}
